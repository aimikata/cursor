  ```python
      """コマンドライン引数からログメッセージを取得する"""
      if len(args) < 2:
          return None
      return " ".join(args[1:])

  def write_log(message):
      """ログメッセージをファイルに書き込む"""
      now = datetime.now()
      log_file = "daily_log.txt"
      with open(log_file, "a", encoding="utf-8") as f:
          f.write(f"{now:%Y-%m-%d %H:%M} {message}\n")
      print(f"Logged: {now:%Y-%m-%d %H:%M} {message}")

  def main():
      """メイン処理"""
      message = get_log_message(sys.argv)
      if message is None:
          print("Usage: python log.py \"やったこと\"")
          print("Example: python log.py \"Chapter3の下書きを書いた\"")
          return
      write_log(message)

  if __name__ == "__main__":
      main()
  ```
  **→ 改善：** `get_log_message()`、`write_log()`、`main()`に分かれている。**各関数の役割が明確でわかりやすい**。
