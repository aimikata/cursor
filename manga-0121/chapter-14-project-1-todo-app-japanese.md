**パネル2：** Composerがプロジェクトを作成する。
- **Composer（提案）：** `todo_app`フォルダを作成し、`todo.py`ファイルを作成する：
  ```python
  # todo.py - ToDoアプリ
  import json
  import os
  from datetime import datetime
  
  DATA_FILE = "todo_data.json"
  
  def load_tasks():
      """タスクを読み込む"""
      if os.path.exists(DATA_FILE):
          with open(DATA_FILE, "r", encoding="utf-8") as f:
              return json.load(f)
      return []
  
  def save_tasks(tasks):
      """タスクを保存する"""
      with open(DATA_FILE, "w", encoding="utf-8") as f:
          json.dump(tasks, f, ensure_ascii=False, indent=2)
  
  def add_task(task):
      """タスクを追加する"""
      tasks = load_tasks()
      task_id = len(tasks) + 1
      tasks.append({
          "id": task_id,
          "task": task,
          "completed": False,
          "created_at": datetime.now().isoformat()
      })
      save_tasks(tasks)
      print(f"タスクを追加しました: {task}")
  
  def delete_task(task_id):
      """タスクを削除する"""
      tasks = load_tasks()
      tasks = [t for t in tasks if t["id"] != task_id]
      save_tasks(tasks)
      print(f"タスクを削除しました: ID {task_id}")
  
  def complete_task(task_id):
      """タスクを完了する"""
      tasks = load_tasks()
      for task in tasks:
          if task["id"] == task_id:
              task["completed"] = True
              break
      save_tasks(tasks)
      print(f"タスクを完了しました: ID {task_id}")
  
  def list_tasks():
      """タスクの一覧を表示する"""
      tasks = load_tasks()
      if not tasks:
          print("タスクがありません")
          return
      for task in tasks:
          status = "✓" if task["completed"] else " "
          print(f"{status} [{task['id']}] {task['task']}")
  
  def main():
      """メイン処理"""
      import sys
      if len(sys.argv) < 2:
          print("Usage: python todo.py [add|delete|complete|list] [args...]")
          return
      command = sys.argv[1]
      if command == "add":
          if len(sys.argv) < 3:
              print("Usage: python todo.py add \"タスク\"")
              return
          add_task(sys.argv[2])
      elif command == "delete":
          if len(sys.argv) < 3:
              print("Usage: python todo.py delete [task_id]")
              return
          delete_task(int(sys.argv[2]))
      elif command == "complete":
          if len(sys.argv) < 3:
              print("Usage: python todo.py complete [task_id]")
              return
          complete_task(int(sys.argv[2]))
      elif command == "list":
          list_tasks()
      else:
          print(f"Unknown command: {command}")
  
  if __name__ == "__main__":
      main()
  ```
